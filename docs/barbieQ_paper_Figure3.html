make_sa_grob <- function(trans, a = 0.35) {
  grid::grid.grabExpr({
    gridGraphics::grid.echo(function() {
      limma::plotSA(
        fit = plot_residuals(mixed, mixed_design, myblock, a = a, trans = trans),
        xlab = "",
        ylab = ""
      )
    })
  })
}
